% 字体自动回退：跨 Windows (MiKTeX/TeX Live) 与 Linux (TeX Live)。
% 不再硬性要求安装 Liberation/Noto 字体；无需随项目附带字体文件。
% !重要：本工程以 XeLaTeX 编译，不能使用 pdfLaTeX。
\usepackage{fontspec}
% 西文字体：Windows 标准字体优先，Linux 常见字体回退。
\IfFontExistsTF{Times New Roman}{\setmainfont{Times New Roman}}{%
  \IfFontExistsTF{Liberation Serif}{\setmainfont{Liberation Serif}}{%
    \IfFontExistsTF{TeXGyreTermes}{\setmainfont{TeXGyreTermes}}{%
      \IfFontExistsTF{DejaVu Serif}{\setmainfont{DejaVu Serif}}{%
        \PackageError{IMUManual}{No Latin serif font found}{Install Times New Roman, Liberation Serif or DejaVu Serif.}}}}}
\IfFontExistsTF{Arial}{\setsansfont{Arial}}{%
  \IfFontExistsTF{Liberation Sans}{\setsansfont{Liberation Sans}}{%
    \IfFontExistsTF{TeXGyreHeros}{\setsansfont{TeXGyreHeros}}{%
      \IfFontExistsTF{DejaVu Sans}{\setsansfont{DejaVu Sans}}{%
        \PackageError{IMUManual}{No Latin sans font found}{Install Arial, Liberation Sans or DejaVu Sans.}}}}}
\IfFontExistsTF{Consolas}{\setmonofont{Consolas}[Scale=0.86]}{%
  \IfFontExistsTF{Liberation Mono}{\setmonofont{Liberation Mono}[Scale=0.86]}{%
    \IfFontExistsTF{DejaVu Sans Mono}{\setmonofont{DejaVu Sans Mono}[Scale=0.86]}{%
      \PackageError{IMUManual}{No Latin mono font found}{Install Consolas, Liberation Mono or DejaVu Sans Mono.}}}}
% 中文字体：Windows 系统字体 -> Linux 常见字体 -> ctex 常见 Fandol 字体。
% AutoFakeBold 防止宋体只有普通字重时大量粗体中文无法正确显示。
\IfFontExistsTF{SimSun}{\setCJKmainfont{SimSun}[AutoFakeBold=2]}{%
  \IfFontExistsTF{Noto Serif CJK SC}{\setCJKmainfont{Noto Serif CJK SC}[AutoFakeBold=2]}{%
    \IfFontExistsTF{FandolSong-Regular}{\setCJKmainfont{FandolSong-Regular}[AutoFakeBold=2]}{%
      \PackageError{IMUManual}{No Chinese serif font found}{Install SimSun, Noto Serif CJK SC or Fandol fonts (ctex).}}}}
\IfFontExistsTF{Microsoft YaHei}{\setCJKsansfont{Microsoft YaHei}[AutoFakeBold=2]}{%
  \IfFontExistsTF{Noto Sans CJK SC}{\setCJKsansfont{Noto Sans CJK SC}[AutoFakeBold=2]}{%
    \IfFontExistsTF{FandolHei-Regular}{\setCJKsansfont{FandolHei-Regular}[AutoFakeBold=2]}{%
      \PackageError{IMUManual}{No Chinese sans font found}{Install Microsoft YaHei, Noto Sans CJK SC or Fandol fonts (ctex).}}}}
\IfFontExistsTF{Microsoft YaHei}{\setCJKmonofont{Microsoft YaHei}[Scale=0.87]}{%
  \IfFontExistsTF{Noto Sans Mono CJK SC}{\setCJKmonofont{Noto Sans Mono CJK SC}[Scale=0.87]}{%
    \IfFontExistsTF{FandolFang-Regular}{\setCJKmonofont{FandolFang-Regular}[Scale=0.87]}{%
      \IfFontExistsTF{FandolHei-Regular}{\setCJKmonofont{FandolHei-Regular}[Scale=0.87]}{}}}}
